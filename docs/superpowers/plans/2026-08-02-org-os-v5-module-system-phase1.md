# org-os v5 Module System — Phase 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the org-os module engine (list/add/adopt/registry), the manifest formats, install-state tracking, and two pilot modules (`org-os-standards`, `org-os-meeting-processor`), per the approved spec `docs/superpowers/specs/2026-08-02-org-os-v5-modularization-design.md` (Phase 1 of §10).

**Architecture:** One engine file, `scripts/modules.mjs`, is the sole reader/writer of `modules/*/module.yaml` (framework registry source), `data/modules.yaml` (instance install manifest), and `.well-known/modules.json` (published state). Modules live in `modules/org-os-<name>/`; installing materializes their files to the canonical instance paths and records per-file sha256 checksums. Phase 1 excludes `update`/`status`/`check` (Phases 2–3).

**Tech Stack:** Node ESM (`.mjs`, repo is `"type": "module"`), `js-yaml` (already a dependency), `node:crypto`, `node:test` built-in runner (no new dependencies).

**Working branch:** `feat/multica-operator` (current). Pre-commit runs `validate:structure` — every commit below must keep it green.

Instance manifest format (`data/modules.yaml`, generated):

```yaml
# Generated by scripts/modules.mjs — do not edit by hand.
modules:
  org-os-standards:
    version: 1.0.0
    installed_at: "2026-08-02T12:00:00.000Z"
    adopted: false
    files:
      scripts/modules.mjs: sha256:ab12…
```

Published state format (`.well-known/modules.json`, generated):

```json
{
  "generated_by": "scripts/modules.mjs",
  "installed": { "org-os-standards": { "version": "1.0.0", "installed_at": "…", "adopted": false } },
  "registry": { "org-os-standards": { "version": "1.0.0", "type": "core", "description": "…", "dependencies": [] } }
}
```

(`registry` appears only where a `modules/` directory exists — i.e. the framework repo.)

```js
#!/usr/bin/env node

/**
 * modules.mjs — org-os module engine (v5, Phase 1)
 *
 * The SOLE reader/writer of module state:
 *   modules/x/module.yaml     — framework registry source
 *   data/modules.yaml         — instance install manifest
 *   .well-known/modules.json  — published state
 *
 * Commands: list | add <id> [--from <framework-path>] | adopt [--from <path>] | registry
 * Phase 2/3 commands (update, status, check) are specified in
 * docs/superpowers/specs/2026-08-02-org-os-v5-modularization-design.md
 */

import { createHash } from 'node:crypto';
import { copyFileSync, existsSync, mkdirSync, readdirSync, readFileSync, writeFileSync } from 'node:fs';
import { dirname, join, resolve } from 'node:path';
import { pathToFileURL } from 'node:url';
import yaml from 'js-yaml';

export const REQUIRED_FIELDS = ['id', 'version', 'type', 'description'];
export const MODULE_TYPES = ['core', 'operational', 'integration'];

export function validateManifest(m) {
  const errors = [];
  if (!m || typeof m !== 'object') return ['manifest is not an object'];
  for (const f of REQUIRED_FIELDS) if (!m[f]) errors.push(`missing required field: ${f}`);
  if (m.id && !/^org-os-[a-z0-9-]+$/.test(m.id)) errors.push(`invalid id: ${m.id}`);
  if (m.version && !/^\d+\.\d+\.\d+$/.test(m.version)) errors.push(`invalid version: ${m.version}`);
  if (m.type && !MODULE_TYPES.includes(m.type)) errors.push(`invalid type: ${m.type}`);
  for (const key of ['files', 'templates']) {
    if (m[key] && (typeof m[key] !== 'object' || Array.isArray(m[key])))
      errors.push(`${key} must be a map of source → target paths`);
  }
  if (m.dependencies && !Array.isArray(m.dependencies)) errors.push('dependencies must be a list');
  return errors;
}
```

```js
export function loadRegistry(frameworkRoot) {
  const modulesDir = join(frameworkRoot, 'modules');
  if (!existsSync(modulesDir)) return {};
  const registry = {};
  for (const entry of readdirSync(modulesDir, { withFileTypes: true })) {
    if (!entry.isDirectory()) continue;
    const manifestPath = join(modulesDir, entry.name, 'module.yaml');
    if (!existsSync(manifestPath)) continue;
    const manifest = yaml.load(readFileSync(manifestPath, 'utf-8'));
    const errors = validateManifest(manifest);
    if (errors.length) throw new Error(`${entry.name}/module.yaml: ${errors.join('; ')}`);
    registry[manifest.id] = { ...manifest, dir: join(modulesDir, entry.name) };
  }
  return registry;
}

export function resolveInstallOrder(registry, id, seen = new Set(), order = []) {
  if (seen.has(id)) return order;
  seen.add(id);
  const mod = registry[id];
  if (!mod) throw new Error(`unknown module: ${id}`);
  for (const dep of mod.dependencies || []) resolveInstallOrder(registry, dep, seen, order);
  order.push(id);
  return order;
}
```

```js
export function sha256(buf) {
  return 'sha256:' + createHash('sha256').update(buf).digest('hex');
}

export function loadInstanceManifest(root) {
  const p = join(root, 'data', 'modules.yaml');
  if (!existsSync(p)) return { modules: {} };
  const parsed = yaml.load(readFileSync(p, 'utf-8'));
  return parsed && typeof parsed === 'object' ? { modules: {}, ...parsed } : { modules: {} };
}

export function writeWellKnown(root, manifest) {
  const out = { generated_by: 'scripts/modules.mjs', installed: {} };
  for (const [id, entry] of Object.entries(manifest.modules || {})) {
    out.installed[id] = {
      version: entry.version,
      installed_at: entry.installed_at,
      adopted: entry.adopted,
    };
  }
  const registry = loadRegistry(root); // {} unless this root is the framework
  if (Object.keys(registry).length) {
    out.registry = Object.fromEntries(
      Object.values(registry).map((m) => [
        m.id,
        { version: m.version, type: m.type, description: m.description, dependencies: m.dependencies || [] },
      ])
    );
  }
  const p = join(root, '.well-known', 'modules.json');
  mkdirSync(dirname(p), { recursive: true });
  writeFileSync(p, JSON.stringify(out, null, 2) + '\n');
}

export function saveInstanceManifest(root, manifest) {
  const p = join(root, 'data', 'modules.yaml');
  mkdirSync(dirname(p), { recursive: true });
  const banner = '# Generated by scripts/modules.mjs — do not edit by hand.\n';
  writeFileSync(p, banner + yaml.dump(manifest, { lineWidth: 100 }));
  writeWellKnown(root, manifest);
}

export function addModule(root, registry, id, { now = new Date().toISOString() } = {}) {
  const manifest = loadInstanceManifest(root);
  const report = [];
  for (const modId of resolveInstallOrder(registry, id)) {
    const mod = registry[modId];
    if (manifest.modules[modId]?.version === mod.version) {
      report.push(`${modId}@${mod.version} already installed`);
      continue;
    }
    const fileState = {};
    for (const [src, target] of Object.entries(mod.files || {})) {
      const targetPath = join(root, target);
      mkdirSync(dirname(targetPath), { recursive: true });
      copyFileSync(join(mod.dir, src), targetPath);
      fileState[target] = sha256(readFileSync(targetPath));
    }
    for (const [src, target] of Object.entries(mod.templates || {})) {
      const targetPath = join(root, target);
      if (existsSync(targetPath)) {
        report.push(`${modId}: template ${target} exists — left untouched`);
        continue;
      }
      mkdirSync(dirname(targetPath), { recursive: true });
      copyFileSync(join(mod.dir, src), targetPath);
      report.push(`${modId}: seeded ${target}`);
    }
    manifest.modules[modId] = { version: mod.version, installed_at: now, adopted: false, files: fileState };
    report.push(`${modId}@${mod.version} installed (${Object.keys(fileState).length} files)`);
  }
  saveInstanceManifest(root, manifest);
  return report;
}
```

```js
export function adoptModules(root, registry, { now = new Date().toISOString() } = {}) {
  const manifest = loadInstanceManifest(root);
  const report = [];
  for (const mod of Object.values(registry)) {
    if (manifest.modules[mod.id]) continue;
    const targets = Object.values(mod.files || {});
    if (!targets.length) continue;
    const present = targets.filter((t) => existsSync(join(root, t)));
    if (!present.length) {
      report.push(`${mod.id}: no files present — skipped`);
      continue;
    }
    const fileState = {};
    for (const t of present) fileState[t] = sha256(readFileSync(join(root, t)));
    manifest.modules[mod.id] = { version: mod.version, installed_at: now, adopted: true, files: fileState };
    report.push(`${mod.id}@${mod.version} adopted (${present.length}/${targets.length} files present)`);
  }
  saveInstanceManifest(root, manifest);
  return report;
}
```

```js
// --- CLI -------------------------------------------------------------------

function flagValue(args, name) {
  const i = args.indexOf(name);
  return i === -1 ? null : args[i + 1];
}

function usage() {
  console.log(`org-os module engine (v5, Phase 1)

Usage:
  npm run module -- list                      List registry vs installed modules
  npm run module -- add <id> [--from <path>]  Install a module (and its dependencies)
  npm run module -- adopt [--from <path>]     Record already-present module files as installed
  npm run module -- registry                  Regenerate .well-known/modules.json

--from points at a framework checkout when running inside an instance
(defaults to the current directory, which is correct in the framework repo).

Phase 2/3 commands (update, status, check) are not yet implemented.`);
}

function main() {
  const [cmd, ...rest] = process.argv.slice(2);
  const root = resolve('.');
  const from = flagValue(rest, '--from') || root;
  try {
    if (cmd === 'list') {
      const registry = loadRegistry(from);
      const manifest = loadInstanceManifest(root);
      const ids = Object.keys(registry).sort();
      if (!ids.length) {
        console.log('No modules found in modules/.');
        return;
      }
      console.log('MODULE'.padEnd(32) + 'LATEST'.padEnd(10) + 'INSTALLED');
      for (const id of ids) {
        const inst = manifest.modules?.[id];
        const state = inst ? `${inst.version}${inst.adopted ? ' (adopted)' : ''}` : '—';
        console.log(id.padEnd(32) + registry[id].version.padEnd(10) + state);
      }
    } else if (cmd === 'add') {
      const id = rest.find((a) => !a.startsWith('--') && a !== flagValue(rest, '--from'));
      if (!id) throw new Error('usage: npm run module -- add <id>');
      for (const line of addModule(root, loadRegistry(from), id)) console.log('  ' + line);
    } else if (cmd === 'adopt') {
      for (const line of adoptModules(root, loadRegistry(from))) console.log('  ' + line);
    } else if (cmd === 'registry') {
      writeWellKnown(root, loadInstanceManifest(root));
```
