```ts
import { existsSync } from 'node:fs'
import { REGISTRIES } from './registries.ts'
import { readRegistry, registryPath, type Entity } from './registry.ts'

export interface GraphNode { id: string; registry: string; label: string; entityId: string; data: Entity }
export interface GraphEdge { source: string; target: string; field: string }
export interface OrgGraph { nodes: GraphNode[]; edges: GraphEdge[] }

const nid = (registry: string, id: string) => `${registry}:${id}`
const labelOf = (e: Entity) => String(e.title ?? e.name ?? e.partner ?? e.id)

/** Cross-references (mirror of schema.ts REFS — the graph is the visual form of the same relations). */
const REFS: { registry: string; field: string; target: string; array?: boolean }[] = [
  { registry: 'projects', field: 'lead', target: 'members' },
  { registry: 'projects', field: 'contributors', target: 'members', array: true },
  { registry: 'projects', field: 'related_ideas', target: 'ideas', array: true },
  { registry: 'projects', field: 'funding_source', target: 'funding-opportunities' },
  { registry: 'meetings', field: 'participants', target: 'members', array: true },
  { registry: 'events', field: 'related_project', target: 'projects' },
]

export function buildOrgGraph(repo: string): OrgGraph {
  const nodes: GraphNode[] = []
  const present = new Set<string>()
  for (const def of REGISTRIES) {
    if (def.kind !== 'collection') continue
    if (!existsSync(registryPath(repo, def.name))) continue
    for (const e of readRegistry(repo, def.name).entries ?? []) {
      const id = nid(def.name, String(e.id))
      nodes.push({ id, registry: def.name, entityId: String(e.id), label: labelOf(e), data: e })
      present.add(id)
    }
  }
  const edges: GraphEdge[] = []
  for (const ref of REFS) {
    if (!existsSync(registryPath(repo, ref.registry))) continue
    for (const e of readRegistry(repo, ref.registry).entries ?? []) {
      const src = nid(ref.registry, String(e.id))
      const v = e[ref.field]
      const targets = ref.array ? (Array.isArray(v) ? v : []) : (v == null ? [] : [v])
      for (const t of targets) {
        const tid = nid(ref.target, String(t))
        if (present.has(tid)) edges.push({ source: src, target: tid, field: ref.field })
      }
    }
  }
  return { nodes, edges }
}
```

### Task 2: Graphify neighborhood expansion

**Files:** Modify `server/graph.ts`; Test append to `tests/graph.test.ts`

On demand, a registry node can expand into its graphify knowledge-graph neighborhood. Read `graphify-out/graph.json` (nodes + edges), find nodes whose id/label references the entity, and return the 1-hop neighborhood. Because graphify output is large and its schema varies, this is best-effort and must degrade gracefully (missing file → empty result).

```ts
import { readFileSync } from 'node:fs'
import { join } from 'node:path'

export interface GraphifyNode { id: string; label: string }
export interface GraphifySub { nodes: GraphifyNode[]; edges: { source: string; target: string }[] }

/** Best-effort 1-hop neighborhood from graphify-out/graph.json. Degrades to empty on any problem. */
export function expandGraphify(repo: string, entityId: string): GraphifySub {
  const path = join(repo, 'graphify-out', 'graph.json')
  let raw: { nodes?: { id: string; label?: string }[]; edges?: { source: string; target: string }[] }
  try { raw = JSON.parse(readFileSync(path, 'utf8')) } catch { return { nodes: [], edges: [] } }
  const allNodes = raw.nodes ?? []
  const allEdges = raw.edges ?? []
  const needle = entityId.toLowerCase()
  const seedIds = new Set(allNodes
    .filter(n => n.id.toLowerCase().includes(needle) || (n.label ?? '').toLowerCase().includes(needle))
    .map(n => n.id))
  if (seedIds.size === 0) return { nodes: [], edges: [] }
  const keep = new Set(seedIds)
  const edges: { source: string; target: string }[] = []
  for (const e of allEdges) {
    if (seedIds.has(e.source)) { keep.add(e.target); edges.push(e) }
    else if (seedIds.has(e.target)) { keep.add(e.source); edges.push(e) }
  }
  const nodes = allNodes.filter(n => keep.has(n.id)).map(n => ({ id: n.id, label: n.label ?? n.id }))
  return { nodes, edges }
}
```

### Task 6: Canvas (Map) component

**Files:** Create `app/src/components/Canvas.tsx`, `app/src/components/NodePanel.tsx`; Modify `App.tsx`, `Sidebar.tsx`

The canvas renders the org graph with a simple force-free layered layout (group by registry into columns), colors nodes by registry, and on node-click opens `NodePanel`. A ⊕ button on the panel calls `graphApi.expand` and injects graphify nodes as a distinct visual layer (dashed, muted). Full React Flow node/edge state is derived from `graphApi.org()`.

- [ ] **Step 1: Write `app/src/components/Canvas.tsx`**

```tsx
import { useMemo, useState } from 'react'
import { ReactFlow, Background, Controls, type Node, type Edge } from '@xyflow/react'
import '@xyflow/react/dist/style.css'
import { useQuery } from '@tanstack/react-query'
import { graphApi, type OrgGraph, type GraphifySub } from '../graph.ts'
import NodePanel from './NodePanel.tsx'

const REGISTRY_ORDER = ['members', 'projects', 'funding-opportunities', 'ideas', 'meetings', 'events', 'relationships']
const colorOf = (r: string) => ({
  members: '#58a6ff', projects: '#7ee787', 'funding-opportunities': '#e3b341',
  ideas: '#d2a8ff', meetings: '#56b6c2', events: '#f0883e', relationships: '#ff7b72',
} as Record<string, string>)[r] ?? '#8b949e'

/** Column-per-registry layout: x by registry index, y by position within registry. */
function layout(g: OrgGraph): { nodes: Node[]; edges: Edge[] } {
  const cols = new Map<string, number>()
  const nodes: Node[] = g.nodes.map(n => {
    const col = REGISTRY_ORDER.indexOf(n.registry)
    const idx = cols.get(n.registry) ?? 0
    cols.set(n.registry, idx + 1)
    return {
      id: n.id, position: { x: (col < 0 ? REGISTRY_ORDER.length : col) * 260, y: idx * 90 },
      data: { label: n.label, node: n },
      style: { border: `1.5px solid ${colorOf(n.registry)}`, borderRadius: 10, background: '#161b22', color: '#c9d1d9', fontSize: 12, padding: 8, width: 200 },
    }
  })
  const edges: Edge[] = g.edges.map((e, i) => ({ id: `e${i}`, source: e.source, target: e.target, style: { stroke: '#30363d' } }))
  return { nodes, edges }
}

export default function Canvas() {
  const { data: graph } = useQuery({ queryKey: ['graph', 'org'], queryFn: graphApi.org })
  const [selected, setSelected] = useState<string | null>(null)
  const [expansion, setExpansion] = useState<GraphifySub | null>(null)
  const base = useMemo(() => (graph ? layout(graph) : { nodes: [], edges: [] }), [graph])

  const expansionNodes: Node[] = (expansion?.nodes ?? []).map((n, i) => ({
    id: `gx:${n.id}`, position: { x: (REGISTRY_ORDER.length + 1) * 260, y: i * 70 },
    data: { label: n.label }, style: { border: '1.5px dashed #8b949e', borderRadius: 10, background: '#0d1117', color: '#8b949e', fontSize: 11, padding: 6, width: 180 },
  }))

  const selectedNode = graph?.nodes.find(n => n.id === selected) ?? null
  return (
    <div style={{ position: 'relative', height: 'calc(100vh - 46px)' }}>
      <ReactFlow
        nodes={[...base.nodes, ...expansionNodes]} edges={base.edges}
        onNodeClick={(_e, n) => { setSelected(n.id); setExpansion(null) }}
        fitView colorMode="dark"
      >
        <Background /><Controls />
      </ReactFlow>
      {selectedNode && (
        <NodePanel node={selectedNode} onClose={() => { setSelected(null); setExpansion(null) }}
          onExpand={async () => setExpansion(await graphApi.expand(selectedNode.registry, selectedNode.entityId))} />
      )}
    </div>
  )
}
```

import { useQuery } from '@tanstack/react-query'
import { api } from '../api.ts'

export default function KanbanView({ source, groupBy }: { source: string; groupBy: string }) {
  const { data: reg } = useQuery({ queryKey: ['registry', source], queryFn: () => api.registry(source) })
  if (!reg?.entries) return <p style={{ color: 'var(--text-dim)' }}>Loading…</p>
  const groups = new Map<string, typeof reg.entries>()
  for (const e of reg.entries) {
    const key = String(e[groupBy] ?? '—')
    ;(groups.get(key) ?? groups.set(key, []).get(key)!).push(e)
  }
  return (
    <div style={{ display: 'flex', gap: 12, overflowX: 'auto' }}>
      {[...groups].map(([key, items]) => (
        <div key={key} style={{ minWidth: 220, background: 'var(--bg-raised)', borderRadius: 8, padding: 10 }}>
          <div className="label" style={{ marginBottom: 8 }}>{key} ({items.length})</div>
          {items.map(e => (
            <div key={e.id} className="card" style={{ marginBottom: 8, padding: 10 }}>
              {String(e.title ?? e.name ?? e.id)}
            </div>
          ))}
        </div>
      ))}
    </div>
  )
}

export default function ViewRenderer({ id }: { id: string }) {

  }
}
